(defvar org-n-label-table nil
  "Hash table of labels to numbers for {{{N(label)}}} references.")

(defun org-n-init-labels ()
  (setq org-n-label-table (make-hash-table :test #'equal))
  (setq org-n-counter 0))

(defun org-n-macro (&optional label)
  "Assigns a number to a label and returns a span with an ID for linking.
If LABEL is nil, tries to find :CUSTOM_ID: property at current heading."
  (let* ((label (or (and (stringp label)
                         (not (string= label ""))
                         (car (split-string label "," t "[ \t\n\r]+")))
                    (org-entry-get nil "CUSTOM_ID")))
         (n (or (gethash label org-n-label-table)
                (puthash label (number-to-string (cl-incf org-n-counter))
                         org-n-label-table))))
    (if label
        (format "<span id=\"%s\">%s</span>" label n)
      (format "??(no label)"))))

(defun org-ref-macro (label)
  "Macro that creates a hyperlink to the numbered label."
  (let ((n (gethash label org-n-label-table)))
    (if n
        (format "<a href=\"#%s\">%s</a>" label n)
      (format "??(%s)" label))))

(defun org-n-html-export-setup (_backend)
  "Prepare label tracking before export."
  (when (eq _backend 'html)
    (org-n-init-labels)))

(add-hook 'org-export-before-processing-hook #'org-n-html-export-setup)

(defvar my-letter-counter 0
  "Counter for the letter macro in Org-mode.")

(defun letter-counter (&optional uppercase)
  "Return the next letter in the sequence.
If UPPERCASE is non-nil, return uppercase letters."
  (let* ((base (if uppercase ?A ?a))
         (letter (char-to-string (+ base my-letter-counter))))
    (setq my-letter-counter (1+ my-letter-counter))
    letter))

(defun reset-letter-counter ()
  "Reset the letter counter to zero."
  (setq my-letter-counter 0)
  "")
